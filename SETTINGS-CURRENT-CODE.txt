                    .size(90.dp)
                    .align(Alignment.CenterHorizontally)
                    .clip(CircleShape),

                tint = MaterialTheme.colorScheme.primary

                style = MaterialTheme.typography.bodySmall

                            .edit()

                            .putString(

                            )
                            .putString(
                                "date_of_birth",
                                dateOfBirth
                            )
                            .putString(
                                "country",

                            )
                            .apply()


                        user?.updateProfile(

                            UserProfileChangeRequest
                                .Builder()
                                .setDisplayName(name)
                                .build()

                        )?.addOnCompleteListener {

                            editing = false
                            saved = true
                        }
                    },

                    modifier = Modifier.fillMaxWidth()
                ) {

                    Text("Save Profile")
                }
            }


            // -----------------------------------------------------
            // SUCCESS MESSAGE
            // -----------------------------------------------------

            if (saved) {

                Text(
                    text = "Profile saved successfully.",
                    color = MaterialTheme.colorScheme.primary
                )
            }
        }

import androidx.compose.runtime.setValue

import com.google.firebase.auth.EmailAuthProvider
import com.google.firebase.auth.FirebaseAuth

@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun SecurityScreen(
    onBack: () -> Unit
) {

    val context = androidx.compose.ui.platform.LocalContext.current

    val preferences = remember {

        context.getSharedPreferences(
            "winx_security",
            Context.MODE_PRIVATE
        )
    }


    var currentPassword by remember {
        mutableStateOf("")
    }

    var newPassword by remember {
        mutableStateOf("")
    }

    var confirmPassword by remember {
        mutableStateOf("")
    }

    var message by remember {
        mutableStateOf("")
    }

    var facialRecognition by remember {

        mutableStateOf(
            preferences.getBoolean(
                "facial_recognition",
                false
            )
        )
    }


    Scaffold(

        topBar = {

            TopAppBar(

                title = {
                    Text("Security")
                },

                navigationIcon = {

                    IconButton(
                        onClick = onBack
                    ) {

                        Icon(
                            imageVector = Icons.Default.ArrowBack,
                            contentDescription = "Back"
                        )
                    }
                }
            )
        }

    ) { paddingValues ->

        Column(

            modifier = Modifier
                .fillMaxSize()
                .padding(paddingValues)
                .padding(20.dp),

            verticalArrangement = Arrangement.spacedBy(12.dp)
        ) {

            Icon(
                imageVector = Icons.Default.Lock,
                contentDescription = "Security"
            )


            Text(
                text = "Password"
            )


            Text(
                text = "Change your password to keep your account secure."
            )


            OutlinedTextField(

                value = currentPassword,

                onValueChange = {
                    currentPassword = it
                },

                label = {
                    Text("Current Password")
                },

                modifier = Modifier.fillMaxWidth()
            )


            OutlinedTextField(

                value = newPassword,

                onValueChange = {
                    newPassword = it
                },

                label = {
                    Text("New Password")
                },

                modifier = Modifier.fillMaxWidth()
            )


            OutlinedTextField(

                value = confirmPassword,

                onValueChange = {
                    confirmPassword = it
                },

                label = {
                    Text("Confirm New Password")
                },

                modifier = Modifier.fillMaxWidth()
            )


            Button(

                onClick = {

                    when {

                        currentPassword.isBlank() -> {

                            message =
                                "Enter your current password."
                        }

                        newPassword.length < 6 -> {

                            message =
                                "New password must contain at least 6 characters."
                        }

                        newPassword != confirmPassword -> {

                            message =
                                "Passwords do not match."
                        }

                        else -> {

                            val user =
                                FirebaseAuth
                                    .getInstance()
                                    .currentUser

                            if (user?.email == null) {

                                message =
                                    "Password changes are only available for email accounts."

                            } else {

                                val credential =
                                    EmailAuthProvider
                                        .getCredential(
                                            user.email!!,
                                            currentPassword
                                        )

                                user
                                    .reauthenticate(credential)
                                    .addOnSuccessListener {

                                        user
                                            .updatePassword(
                                                newPassword
                                            )
                                            .addOnSuccessListener {

                                                currentPassword = ""
                                                newPassword = ""
                                                confirmPassword = ""

                                                message =
                                                    "Password updated successfully."
                                            }
                                            .addOnFailureListener {

                                                message =
                                                    it.message
                                                        ?: "Unable to update password."
                                            }
                                    }
                                    .addOnFailureListener {

                                        message =
                                            "Current password is incorrect."
                                    }
                            }
                        }
                    }
                },

                modifier = Modifier.fillMaxWidth()
            ) {

                Text("Update Password")
            }


            if (message.isNotBlank()) {

                Text(message)
            }


            // -----------------------------------------------------
            // FACIAL RECOGNITION PREFERENCE
            // -----------------------------------------------------

            Row(

                modifier = Modifier.fillMaxWidth(),

                verticalAlignment =
                    Alignment.CenterVertically
            ) {

                Icon(
                    imageVector = Icons.Default.Face,
                    contentDescription = "Facial Recognition"
                )


                Text(

                    text = "Facial Recognition",

                    modifier = Modifier
                        .weight(1f)
                        .padding(start = 12.dp)
                )


                Switch(

                    checked = facialRecognition,

                    onCheckedChange = {

                        facialRecognition = it

                        preferences
                            .edit()
                            .putBoolean(
                                "facial_recognition",
                                it
                            )
                            .apply()
                    }
                )
            }


            Text(

                text = if (facialRecognition) {

                    "Facial recognition preference enabled."

                } else {

                    "Facial recognition is not enabled."
                }
            )
        }
    }
}
